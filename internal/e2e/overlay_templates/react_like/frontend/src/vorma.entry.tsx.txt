import { createRoot } from "react-dom/client";
import {
	getRootEl,
	initClient,
} from "vorma/client";
import { VormaRootOutlet } from "vorma/react";
import { installE2EBridge } from "./vorma.bindings.ts";
import { vormaAppConfig } from "./vorma.gen/index.ts";

		root.render(<VormaRootOutlet />);
	},
});
