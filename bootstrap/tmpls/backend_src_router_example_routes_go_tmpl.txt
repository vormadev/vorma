package router

import "github.com/vormadev/vorma"

type RootData struct {
	Message string
}

const externalDocsLink = "https://vorma.dev/docs"

var count = 0

func registerExampleLoaders() {
	NewLoader("/", func(c *LoaderCtx) (*RootData, error) {
		return &RootData{Message: "Welcome to your Vorma app!"}, nil
	})

	NewLoader("/_index", func(c *LoaderCtx) (int, error) {
		return count, nil
	})

	NewLoader("/links", func(c *LoaderCtx) (string, error) {
		return externalDocsLink, nil
	})
}

func registerExampleActions() {
	NewAction("POST", "/increment-count", func(c *ActionCtx[vorma.None]) (int, error) {
		count++
		return count, nil
	})
}
